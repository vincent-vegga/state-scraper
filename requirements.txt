# Descarga de los feeds por HTTPS, con timeouts y reintentos.
requests>=2.32,<3

# Parseo de XML. Se elige lxml frente a BeautifulSoup porque
# permite XPath con local-name(), que localiza las etiquetas
# CODICE (cac:, cbc:, cac-place-ext:) sin declarar un mapa de
# namespaces que quedaria obsoleto en cada version del esquema.
lxml>=5.2,<7

# Interpretacion robusta de fechas ISO 8601 con zonas horarias
# y formatos ligeramente irregulares entre publicadores.
python-dateutil>=2.9,<3

# --- Almacen comun ------------------------------------------

# Cliente oficial de Supabase: unico destino de los datos.
supabase>=2.10,<3
